SHA256SUMS.txtは各ファイルの指紋一覧です。コピー間違い・破損の検出用で、第三者による真正性の保証ではありません。
展開先で次を実行すると確認できます。

~~~powershell
$failed = $false
Get-Content -LiteralPath .\SHA256SUMS.txt -Encoding UTF8 | ForEach-Object {
    $hash, $relative = $_ -split '  ', 2
    $actual = (Get-FileHash -LiteralPath $relative -Algorithm SHA256).Hash
    if ($actual -ne $hash) {
        $failed = $true
        Write-Host "STOP: $relative"
    }
}
if ($failed) { throw 'ファイルが一致しません。取り込みを止めてください。' }
Write-Host 'PASS: 全ファイルが一致しました'
~~~
